//@version=5
strategy("境界當沖｜McGinley Dynamic反手+4口分層停利【診斷版】",
     overlay=true,
     initial_capital=1000000,
     default_qty_type=strategy.fixed,
     default_qty_value=4,
     pyramiding=0,
     commission_type=strategy.commission.cash_per_order,
     commission_value=50,
     process_orders_on_close=false,
     max_labels_count=500,
     max_lines_count=500)

startTime = input.time(timestamp("2024-01-01T00:00:00"), "回測開始")
endTime   = input.time(timestamp("2030-01-01T00:00:00"), "回測結束(預設涵蓋至今)")
inRange = time >= startTime and time <= endTime

mdLen = input.int(14, "McGinley Dynamic週期")
var float md = na
md := na(md[1]) ? close : md[1] + (close - md[1]) / (mdLen * math.pow(close / md[1], 4))

simpleLabel = input.bool(true, "簡化標籤文字(只顯示簡短符號,不顯示完整說明)")
cmtLongEntry  = simpleLabel ? "+4" : "多單進場(4口)"
cmtShortEntry = simpleLabel ? "-4" : "空單進場(4口)"
cmtLongExit   = simpleLabel ? "平多進空" : "平多翻空"
cmtShortExit  = simpleLabel ? "平空進多" : "平空翻多"
cmtTP1        = simpleLabel ? "TP1" : "第1口停利"
cmtTP2        = simpleLabel ? "TP2" : "第2口停利"
cmtTP3        = simpleLabel ? "TP3" : "第3口停利"

goldenFlip = ta.crossover(close, md)
deathFlip  = ta.crossunder(close, md)

useBuffer = input.bool(true, "啟用緩衝帶(過濾盤整雜訊號)")
bufferAtrLen = input.int(14, "緩衝帶ATR週期")
bufferAtrMult = input.float(0.15, "緩衝帶ATR倍數(越大越不敏感)")
bufferAmt = ta.atr(bufferAtrLen) * bufferAtrMult

goldenFlipConfirmed = useBuffer ? (close > md + bufferAmt and close[1] <= md[1] + bufferAmt) : goldenFlip
deathFlipConfirmed  = useBuffer ? (close < md - bufferAmt and close[1] >= md[1] - bufferAmt) : deathFlip

if inRange and goldenFlipConfirmed
    strategy.close("Short", comment=cmtShortExit)
    strategy.entry("Long", strategy.long, qty=4, comment=cmtLongEntry)

if inRange and deathFlipConfirmed
    strategy.close("Long", comment=cmtLongExit)
    strategy.entry("Short", strategy.short, qty=4, comment=cmtShortEntry)

var float fEntryPrice = na

if inRange and goldenFlipConfirmed
    fEntryPrice := close

if inRange and deathFlipConfirmed
    fEntryPrice := close

if strategy.position_size == 0
    fEntryPrice := na

tp1Points = input.float(150, "第1口停利點數")
tp2Points = input.float(330, "第2口停利點數")
tp3Points = input.float(600, "第3口停利點數")

tp1Long = fEntryPrice + tp1Points
tp1Short = fEntryPrice - tp1Points
tp2Long = fEntryPrice + tp2Points
tp2Short = fEntryPrice - tp2Points
tp3Long = fEntryPrice + tp3Points
tp3Short = fEntryPrice - tp3Points

if strategy.position_size > 0 and not na(fEntryPrice)
    strategy.exit("多方第1口", from_entry="Long", qty=1, limit=tp1Long, comment=cmtTP1)
    strategy.exit("多方第2口", from_entry="Long", qty=1, limit=tp2Long, comment=cmtTP2)
    strategy.exit("多方第3口", from_entry="Long", qty=1, limit=tp3Long, comment=cmtTP3)

if strategy.position_size < 0 and not na(fEntryPrice)
    strategy.exit("空方第1口", from_entry="Short", qty=1, limit=tp1Short, comment=cmtTP1)
    strategy.exit("空方第2口", from_entry="Short", qty=1, limit=tp2Short, comment=cmtTP2)
    strategy.exit("空方第3口", from_entry="Short", qty=1, limit=tp3Short, comment=cmtTP3)

showBufferLines = input.bool(false, "顯示緩衝帶輔助線(僅供除錯用)")
plot(md, "McGinley Dynamic", color=close > md ? color.lime : color.red, linewidth=2)
plot(useBuffer and showBufferLines ? md + bufferAmt : na, "緩衝帶上界", color=color.new(color.orange, 30), style=plot.style_circles, linewidth=2)
plot(useBuffer and showBufferLines ? md - bufferAmt : na, "緩衝帶下界", color=color.new(color.orange, 30), style=plot.style_circles, linewidth=2)

modeText = close > md ? "目前多方" : "目前空方"
currentText =
     strategy.position_size > 0 ? "持有多單(" + str.tostring(int(strategy.position_size)) + "口)" :
     strategy.position_size < 0 ? "持有空單(" + str.tostring(int(math.abs(strategy.position_size))) + "口)" :
     "空手(等待下一次翻轉,或已被停利出場)"

totalTrades = strategy.closedtrades
winTrades = strategy.wintrades
lossTrades = strategy.losstrades
grossProfit = strategy.grossprofit
grossLoss = math.abs(strategy.grossloss)
winRate = totalTrades > 0 ? winTrades * 100.0 / totalTrades : 0.0
pf = grossLoss > 0 ? grossProfit / grossLoss : na
avgWin = winTrades > 0 ? grossProfit / winTrades : 0.0
avgLoss = lossTrades > 0 ? grossLoss / lossTrades : 0.0
rr = avgLoss > 0 ? avgWin / avgLoss : na

showPanel = input.bool(true, "顯示右上操盤面板")
var table panel = table.new(position.top_right, 1, 10, border_width=0, frame_width=0)

if barstate.islast and showPanel
    bgDark = color.new(color.black, 12)
    sepCol = color.new(color.gray, 60)
    panelColor = strategy.position_size > 0 ? color.green : strategy.position_size < 0 ? color.red : color.gray

    table.cell(panel, 0, 0, "McGinley Dynamic反手｜4口分層停利【診斷版】", text_color=color.white, bgcolor=panelColor, text_size=size.normal, text_halign=text.align_left)
    table.cell(panel, 0, 1, "━━━━━━━━━━━━━━", text_color=sepCol, bgcolor=bgDark, text_size=size.tiny, text_halign=text.align_left)
    table.cell(panel, 0, 2, "模式:" + modeText, text_color=color.white, bgcolor=bgDark, text_halign=text.align_left)
    table.cell(panel, 0, 3, "狀態:" + currentText, text_color=color.yellow, bgcolor=bgDark, text_halign=text.align_left)
    table.cell(panel, 0, 4, "━━━━━━━━━━━━━━", text_color=sepCol, bgcolor=bgDark, text_size=size.tiny, text_halign=text.align_left)
    table.cell(panel, 0, 5, "交易:" + str.tostring(totalTrades) + "　勝率:" + str.tostring(winRate, "#.##") + "%", text_color=color.white, bgcolor=bgDark, text_halign=text.align_left)
    table.cell(panel, 0, 6, "PF:" + (na(pf) ? "--" : str.tostring(pf, "#.##")) + "　盈虧比:" + (na(rr) ? "--" : str.tostring(rr, "#.##")), text_color=color.white, bgcolor=bgDark, text_halign=text.align_left)
    table.cell(panel, 0, 7, "最大回撤:" + str.tostring(strategy.max_drawdown, "#"), text_color=color.white, bgcolor=bgDark, text_halign=text.align_left)
    table.cell(panel, 0, 8, "總損益:" + str.tostring(strategy.netprofit, "#"), text_color=color.white, bgcolor=bgDark, text_halign=text.align_left)
